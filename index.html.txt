  <script>
    let members = [];

    function normalizeName(text) {
      return text.trim().replace(/\s+/g, "");
    }

    function normalizeHeight(text) {
      text = text.trim()
        .replace("㎝", "cm")
        .replace("ＣＭ", "cm")
        .replace("ｃｍ", "cm");

      if (!text) return "";

      if (text === "不明") return "不明";

      if (/^\d+(\.\d+)?$/.test(text)) {
        return text + "cm";
      }

      return text;
    }

    function normalizeBloodType(text) {
      text = text.trim().toUpperCase();

      if (!text) return "";

      if (text === "不明") return "不明";

      if (!text.endsWith("型")) {
        text += "型";
      }

      return text;
    }

    function parseMembers(text) {
      const lines = text.split(/\r?\n/);
      const parsed = [];

      for (const line of lines) {
        const trimmedLine = line.trim();

        if (!trimmedLine) continue;
        if (!trimmedLine.includes("：")) continue;

        const [name, profile] = trimmedLine.split("：");
        const parts = profile.split("／").map(part => part.trim());

        if (parts.length !== 5) continue;

        const [birthday, zodiac, height, birthplace, bloodType] = parts;

        parsed.push({
          name: name.trim(),
          birthday,
          zodiac,
          height,
          birthplace,
          bloodType
        });
      }

      return parsed;
    }

    async function loadMembers() {
      try {
        const response = await fetch("members.txt");

        if (!response.ok) {
          throw new Error("members.txt を読み込めませんでした。");
        }

        const text = await response.text();
        members = parseMembers(text);

        document.getElementById("status").textContent =
          `${members.length}人分のデータを読み込みました。検索条件を入力してください。`;

      } catch (error) {
        document.getElementById("status").textContent =
          "データの読み込みに失敗しました。index.html と members.txt が同じフォルダにあるか確認してください。";
      }
    }

    function searchMembers() {
      const name = normalizeName(document.getElementById("nameInput").value);
      const birthday = document.getElementById("birthdayInput").value.trim();
      const zodiac = document.getElementById("zodiacInput").value.trim();
      const height = normalizeHeight(document.getElementById("heightInput").value);
      const birthplace = document.getElementById("birthplaceInput").value.trim();
      const bloodType = normalizeBloodType(document.getElementById("bloodTypeInput").value);

      const results = members.filter(member => {
        if (name && !normalizeName(member.name).includes(name)) {
          return false;
        }

        // 例：「2002」「9月7日」「2002年9月7日」のどれでも検索可能
        if (birthday && !member.birthday.includes(birthday)) {
          return false;
        }

        if (zodiac && member.zodiac !== zodiac) {
          return false;
        }

        if (height && normalizeHeight(member.height) !== height) {
          return false;
        }

        // 例：「大阪」でも「大阪府」にヒット
        if (birthplace && !member.birthplace.includes(birthplace)) {
          return false;
        }

        if (bloodType && normalizeBloodType(member.bloodType) !== bloodType) {
          return false;
        }

        return true;
      });

      showResults(results);
    }

    function showResults(results) {
      const status = document.getElementById("status");
      const table = document.getElementById("resultTable");
      const tbody = document.getElementById("resultBody");

      tbody.innerHTML = "";

      if (results.length === 0) {
        status.textContent = "該当するメンバーはいません。";
        table.style.display = "none";
        return;
      }

      status.textContent = `${results.length}件見つかりました。`;
      table.style.display = "table";

      for (const member of results) {
        const tr = document.createElement("tr");

        tr.innerHTML = `
          <td>${escapeHtml(member.name)}</td>
          <td>${escapeHtml(member.birthday)}</td>
          <td>${escapeHtml(member.zodiac)}</td>
          <td>${escapeHtml(member.height)}</td>
          <td>${escapeHtml(member.birthplace)}</td>
          <td>${escapeHtml(member.bloodType)}</td>
        `;

        tbody.appendChild(tr);
      }
    }

    function resetSearch() {
      document.getElementById("nameInput").value = "";
      document.getElementById("birthdayInput").value = "";
      document.getElementById("zodiacInput").value = "";
      document.getElementById("heightInput").value = "";
      document.getElementById("birthplaceInput").value = "";
      document.getElementById("bloodTypeInput").value = "";

      document.getElementById("resultTable").style.display = "none";
      document.getElementById("status").textContent =
        `${members.length}人分のデータを読み込みました。検索条件を入力してください。`;
    }

    function escapeHtml(text) {
      return String(text)
        .replaceAll("&", "&amp;")
        .replaceAll("<", "&lt;")
        .replaceAll(">", "&gt;")
        .replaceAll('"', "&quot;")
        .replaceAll("'", "&#039;");
    }

    document.getElementById("searchButton").addEventListener("click", searchMembers);
    document.getElementById("resetButton").addEventListener("click", resetSearch);

    loadMembers();
  </script>
